\chapter{Objetivos Primários e Secundários}

\section{Objetivos Primários}

\begin{itemize}
    \item Conceber e desenvolver o \textbf{Cognitivo}, um software gamificado educacional que opere os quatro pilares do Pensamento Computacional de forma sequencial e obrigatória na resolução de questões reais de matemática do ENEM;
    \item Avaliar a experiência e o engajamento dos estudantes por meio de instrumentos validados (ESM e UES).
\end{itemize}

\section{Objetivos Secundários}

\begin{itemize}
    \item Mapear as principais competências matemáticas das questões do ENEM e identificar como os pilares do PC são mobilizados em cada uma delas;
    \item Implementar um pipeline ETL para extração, tratamento e enriquecimento de questões reais do ENEM a partir da API ENEM DEV (\textit{self-hosted});
    \item Projetar a arquitetura do sistema com separação de responsabilidades (Plataforma Educacional Gamificada e Módulo ETL);
    \item Conceber e validar diagramas de caso de uso e fluxo de navegabilidade da plataforma;
    \item Implementar mecânicas de gamificação (XP Geral, XP de Atributo, insígnias, ligas, matriz de penalidades e custo estratégico de dicas);
    \item Validar o artefato educacional com professores de matemática por meio de grupo focal (Ciclo 1);
    \item Validar o sistema com alunos da UNEB, aplicando os formulários ESM e UES (Ciclo 2).
\end{itemize}
