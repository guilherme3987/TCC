\chapter{Desfecho Primário e Secundário}

\section{Desfecho Primário}

[Disponibilização do sistema Cognitivo, um produto de \textit{software} de apoio ao ensino de Matemática no Ensino Médio fundamentado nos pilares do Pensamento Computacional, acompanhada do nível de engajamento dos estudantes com a plataforma, mensurado pela Escala de Engajamento do Usuário (UES), considerando as quatro dimensões: Atenção Focada, Usabilidade Percebida, Apelo Estético e Recompensa.]

\section{Desfecho Secundário}

\begin{itemize}
    \item{} [Construção de evidências sobre a relação entre a Matemática e o desenvolvimento do Pensamento Computacional.]
    \item{} [Percepção dos estudantes sobre a utilidade do Pensamento Computacional para a resolução de problemas matemáticos do ENEM.]
\end{itemize}
