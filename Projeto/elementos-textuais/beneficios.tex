\chapter{Benefícios}

\section{Benefícios Diretos para os Participantes}

\begin{itemize}
    \item Estimula o engajamento, o raciocínio lógico e a autonomia na resolução de problemas de Matemática do \gls{enem}, promovendo o desenvolvimento do Pensamento Computacional e incentivando a persistência por meio de mecânicas gamificadas.
    \item Proporciona a oportunidade de colaborar na validação formativa de uma tecnologia educacional, avaliando a acurácia dos conteúdos e a aplicabilidade pedagógica da ferramenta no contexto escolar.
\end{itemize}

\section{Benefícios Indiretos para a Sociedade}

\begin{itemize}
    \item Contribuição para o campo da Informática na Educação, com documentação do processo de desenvolvimento e validação.
    \item Geração de dados educacionais que podem alimentar análises de mineração de dados educacionais em trabalhos futuros.
    \item Modelo replicável de integração entre Pensamento Computacional e gamificação.
\end{itemize}

\section{Forma de Aplicação dos Benefícios}

\begin{itemize}
    \item Disponibilização do sistema Cognitivo para uso dos participantes durante a pesquisa.
\end{itemize}
